```go
// RefreshGeometry 几何静默校准（Phase 0）：末次移动/缩放事件可能滞后于窗口真实
// 几何，静默期后回读补记。位置与尺寸均无变化时不动作（不刷 UpdatedAt/不落盘）。
func (c *Controller) RefreshGeometry(id string, x, y, w, h int) {
	c.mu.Lock()
	n := c.findLocked(id)
	if n == nil {
		c.mu.Unlock()
		return
	}
	moved := n.px != x || n.py != y
	resized := model.ClampW(n.rec.W) != model.ClampW(w) || model.ClampH(n.rec.H) != model.ClampH(h)
	if !moved && !resized {
		c.mu.Unlock()
		return
	}
	if moved {
		n.px, n.py = x, y
		n.rec.X, n.rec.Y = x, y // 与 NoteMoved 同语义：视为用户摆放
	}
	if resized {
		n.rec.W, n.rec.H = w, h
	}
	n.rec.UpdatedAt = time.Now().UnixMilli()
	c.mu.Unlock()
	c.scheduleSave()
}
```

```go
package wm

// pushGap 弹开落点与遮挡者之间保留的间隙（与装箱边距同源）。
const pushGap = PlaceMargin

// 弹开动画节奏（GlideNote 参数，wails 侧按 ms 步进）。
const (
	GlideMs        = 250
	GlideStaggerMs = 40
)

// PushNote 一张候选被弹便签的输入（实时几何：px/py 优先 + 缩放后 ClampW/H）。
type PushNote struct {
	ID   string
	Rect Rect
}

// PushMode 被弹者解析方式。
type PushMode int

const (
	PushSlide PushMode = iota // 方向弹开（滑到目标位）
	PushPack                  // 四向不可行，装箱兜底找空位
)

// Push 一张被弹便签的解析结果（仅含实际移动者；接受重叠者不在结果中）。
type Push struct {
	ID   string
	X, Y int
	Mode PushMode
}

// ResolveOverlaps 召唤弹开核心（纯函数）。self 为已落地召唤便签矩形；notes 为
// 其余全部可见便签（实时几何，皆可推）。与「已落地矩形集」相交的便签被弹开：
// 右/下/左/上迭代越过已落地遮挡者（迭代有界），取可行者中位移最小者（平手偏向
// 右/下——候选按此顺序生成、严格小于才替换），落点留 pushGap 间隙且完整在屏内
//（PlaceMargin 内缩）；四向均不可行 → 装箱对全部其他便签避让+复核；仍无空位 →
// 接受重叠（原地不动、不出现在结果）。被弹者落点撞上未处理便签时，该便签在其
// 后的扫描轮被弹开（级联）；每张至多解析一次，循环有界。已落地矩形两两无交
//（归纳：每次落点都对当时全部已落地者复核），推进迭代必然终止。
func ResolveOverlaps(self Rect, notes []PushNote, screenW, screenH float32) []Push {
	landed := []Rect{self}
	byID := make(map[string]Rect, len(notes))
	for _, n := range notes {
		byID[n.ID] = n.Rect
	}
	processed := map[string]bool{}
	out := make([]Push, 0, len(notes))

	for {
		progressed := false
		for _, n := range notes {
			if processed[n.ID] {
				continue
			}
			cur := byID[n.ID]
			if !overlapsAny(cur, landed) {
				continue
			}
			processed[n.ID] = true
			bestDist := float32(-1)
			var bestX, bestY float32
			for _, d := range slideDisps(cur, landed, screenW, screenH) {
				dist := absF(d.dx) + absF(d.dy)
				if bestDist < 0 || dist < bestDist {
					bestDist, bestX, bestY = dist, d.dx, d.dy
				}
			}
			if bestDist >= 0 {
				nx, ny := int(cur.X+bestX), int(cur.Y+bestY)
				nr := Rect{X: float32(nx), Y: float32(ny), W: cur.W, H: cur.H}
				landed = append(landed, nr)
				byID[n.ID] = nr
				out = append(out, Push{ID: n.ID, X: nx, Y: ny, Mode: PushSlide})
				progressed = true
				continue
			}
			// 装箱兜底：对全部其他便签（含 self）当前矩形避让 + 复核
			rects := make([]Rect, 0, len(notes)+1)
			rects = append(rects, self)
			for _, o := range notes {
				if o.ID != n.ID {
					rects = append(rects, byID[o.ID])
				}
			}
			pnx, pny := PackTopLeft(rects, cur.W, cur.H, screenW, screenH)
			nr := Rect{X: pnx, Y: pny, W: cur.W, H: cur.H}
			if !overlapsAny(nr, rects) {
				landed = append(landed, nr)
				byID[n.ID] = nr
				out = append(out, Push{ID: n.ID, X: int(pnx), Y: int(pny), Mode: PushPack})
				progressed = true
			}
			// 否则接受重叠：原地不动
		}
		if !progressed {
			break
		}
	}
	return out
}

type disp struct{ dx, dy float32 }

// slideDisps 四方向候选（右/下/左/上顺序，供平手偏向右/下），只返回可行者。
func slideDisps(r Rect, landed []Rect, screenW, screenH float32) []disp {
	const m = PlaceMargin
	var out []disp
	if dx, ok := advanceX(r, landed, screenW-m, true); ok {
		out = append(out, disp{dx: dx})
	}
	if dy, ok := advanceY(r, landed, screenH-m, true); ok {
		out = append(out, disp{dy: dy})
	}
	if dx, ok := advanceX(r, landed, m, false); ok {
		out = append(out, disp{dx: dx})
	}
	if dy, ok := advanceY(r, landed, m, false); ok {
		out = append(out, disp{dy: dy})
	}
	return out
}

// advanceX 水平迭代推进：不断越过相交的已落地矩形直到候选无交；出界不可行。
// toRight=true 向右（limit=右界），否则向左（limit=左界）。返回位移量。
func advanceX(r Rect, landed []Rect, limit float32, toRight bool) (float32, bool) {
	x := r.X
	for i := 0; i <= len(landed); i++ {
		cand := Rect{X: x, Y: r.Y, W: r.W, H: r.H}
		hit := false
		for _, l := range landed {
			if rectsIntersect(cand, l) {
				if toRight {
					x = l.X + l.W + pushGap
				} else {
					x = l.X - pushGap - r.W
				}
				hit = true
				break
			}
		}
		if !hit {
			if toRight {
				if x+r.W <= limit {
					return x - r.X, true
				}
			} else if x >= limit {
				return x - r.X, true
			}
			return 0, false
		}
	}
	return 0, false
}

// advanceY 垂直迭代推进（同 advanceX 语义）。
func advanceY(r Rect, landed []Rect, limit float32, toDown bool) (float32, bool) {
	y := r.Y
	for i := 0; i <= len(landed); i++ {
		cand := Rect{X: r.X, Y: y, W: r.W, H: r.H}
		hit := false
		for _, l := range landed {
			if rectsIntersect(cand, l) {
				if toDown {
					y = l.Y + l.H + pushGap
				} else {
					y = l.Y - pushGap - r.H
				}
				hit = true
				break
			}
		}
		if !hit {
			if toDown {
				if y+r.H <= limit {
					return y - r.Y, true
				}
			} else if y >= limit {
				return y - r.Y, true
			}
			return 0, false
		}
	}
	return 0, false
}

func overlapsAny(r Rect, rects []Rect) bool {
	for _, o := range rects {
		if rectsIntersect(r, o) {
			return true
		}
	}
	return false
}

func absF(v float32) float32 {
	if v < 0 {
		return -v
	}
	return v
}
```

```go
func (c *Controller) summonNote(id string, focus bool) {

// pushPass 落地便签的弹开趟：收集其余可见便签实时几何 → ResolveOverlaps 解析
//（级联/兜底）→ 状态先写（px/py；placed 同步 rec.X/Y、哨兵不动、不刷 UpdatedAt）
// → GlideNote 动画视觉追赶（瀑布错峰）。整趟一次防抖落盘。
func (c *Controller) pushPass(self *NoteLive) {
	c.mu.Lock()
	selfRect := Rect{
		X: float32(self.px), Y: float32(self.py),
		W: float32(model.ClampW(self.rec.W)), H: float32(model.ClampH(self.rec.H)),
	}
	notes := make([]PushNote, 0, len(c.notes))
	lives := map[string]*NoteLive{}
	for _, n := range c.notes {
		if n.rec.ID == self.rec.ID || !c.visible[n.rec.ID] {
			continue
		}
		sx, sy, ok := originLocked(n)
		if !ok {
			continue
		}
		notes = append(notes, PushNote{
			ID: n.rec.ID,
			Rect: Rect{
				X: float32(sx), Y: float32(sy),
				W: float32(model.ClampW(n.rec.W)), H: float32(model.ClampH(n.rec.H)),
			},
		})
		lives[n.rec.ID] = n
	}
	c.mu.Unlock()
	if len(notes) == 0 {
		return
	}
	sw, sh := c.ops.PrimaryScreen()
	pushes := ResolveOverlaps(selfRect, notes, sw, sh)
	if len(pushes) == 0 {
		return
	}
	for i, p := range pushes {
		n := lives[p.ID]
		c.mu.Lock()
		n.px, n.py = p.X, p.Y
		if n.rec.X != model.SentinelCoord && n.rec.Y != model.SentinelCoord {
			n.rec.X, n.rec.Y = p.X, p.Y // 弹后位置记住（placed）；哨兵保持哨兵
		}
		c.mu.Unlock()
		c.ops.GlideNote(p.ID, p.X, p.Y, i*GlideStaggerMs, GlideMs)
	}
	c.scheduleSave()
	log.Printf("[summon] %s 落地原位，弹开 %d 张", self.rec.ID, len(pushes))
}
```

```go
const (
	glideStepMs = 16
	geomSettleMs = 150
)

// GlideNote 弹开动画：步进 tween（easeOutCubic），全程抑制 WindowDidMove 记账；
// 同窗口再次调用 close 旧取消信号取旧起新。锁只护 map 访问，SetPosition 在锁外。
func (o *wailsOps) GlideNote(id string, x, y, delayMs, ms int) {
	o.animMu.Lock()
	if done, ok := o.animDone[id]; ok {
		close(done)
	}
	done := make(chan struct{})
	o.animDone[id] = done
	o.suppressMove[id] = true
	o.animMu.Unlock()

	go func() {
		if delayMs > 0 {
			select {
			case <-done:
				o.finishGlide(id, done)
				return
			case <-time.After(time.Duration(delayMs) * time.Millisecond):
			}
		}
		win := o.lookup(id)
		if win == nil {
			o.finishGlide(id, done)
			return
		}
		sx, sy := win.Position()
		steps := ms / glideStepMs
		if steps < 1 {
			steps = 1
		}
		for i := 1; i <= steps; i++ {
			select {
			case <-done:
				o.finishGlide(id, done)
				return
			default:
			}
			t := float64(i) / float64(steps)
			e := 1 - (1-t)*(1-t)*(1-t) // easeOutCubic
			nx := sx + int(math.Round(float64(x-sx)*e))
			ny := sy + int(math.Round(float64(y-sy)*e))
			win.SetPosition(nx, ny)
			time.Sleep(glideStepMs * time.Millisecond)
		}
		win.SetPosition(x, y) // 精确落点
		o.finishGlide(id, done)
	}()
}

func (o *wailsOps) finishGlide(id string, done chan struct{}) {
	o.animMu.Lock()
	if o.animDone[id] == done { // 仍是本动画持有标记才清理（被新动画接管则不动）
		delete(o.animDone, id)
		delete(o.suppressMove, id)
	}
	o.animMu.Unlock()
}
```
