> **git 纪律（工程硬性规则 1）**：所有 git 操作由开发者本人执行，AI 不执行、本计划不含 git 命令步骤；每个 Task 完成后提示开发者可自行提交。

**Goal:** 拖动/缩放便签 A 的手势过程中，被 A 压到的可见便签实时动画弹开（级联、装箱兜底、屏幕满接受重叠），便签之间永不重叠；松手/键盘移动等会话外路径由记账兜底覆盖。

**Architecture:** 框架的 `WindowDidMove` 是 50ms 尾部去抖（拖动中不触发），故实时挤开走**手势会话采样**：订阅 `WindowStartMove/EndMove`、`WindowStartResize/EndResize`，Start 起 50ms 采样 goroutine → `Controller.DragLiveTick`（只写实时几何 px/py/rw/rh）→ 复用 v3.0.0 `pushPass`/`ResolveOverlaps` 弹开；`NoteMoved/NoteResized` 记账后补弹开趟兜底（松手残留、拖动中停顿、Win+方向键键盘移动）。框架零改动。

**Tech Stack:** Go（Wails v3 后端）、go test 单测（fakeOps 内存替身）。

**Spec:** `docs/project/便签/2026-11-19-v3.1.0-drag-push-away-design.md`

```go
// NoteLive 一张活动便签：权威记录 + 最近实际摆放位置。
// rec.X/Y 保持 -1 哨兵直到用户拖动；装箱避让需要兄弟便签的实时位置——
// px/py 即该实时值（对应 bugzia「读兄弟窗口实时几何」的进程内等效）。
type NoteLive struct {
	rec model.NoteRecord
	px  int
	py  int
	// rw/rh 缩放手势中的实时尺寸（v3.1.0）；0 = 未置，回退 ClampW/H(rec.W/H)。
	rw int
	rh int
}
```

```go
// liveSizeLocked 便签实时尺寸：缩放会话值优先，回退 Clamp 后的记录值。须持锁。
func liveSizeLocked(n *NoteLive) (int, int) {
	if n.rw > 0 && n.rh > 0 {
		return n.rw, n.rh
	}
	return model.ClampW(n.rec.W), model.ClampH(n.rec.H)
}
```

```go
// DragLiveTick 拖动/缩放手势采样（v3.1.0 实时挤开）：只写实时几何（px/py 与
// rw/rh），随后 pushPass 实时弹开被压便签。对被操作便签不写 rec、不刷
// UpdatedAt、不落盘——权威记账仍归松手后的 NoteMoved/NoteResized；
// pushPass 弹开被压者时照常记账落盘（弹开位置是持久事实，语义同 v3.0.0）。
func (c *Controller) DragLiveTick(id string, x, y, w, h int) {
	c.mu.Lock()
	n := c.findLocked(id)
	if n == nil {
		c.mu.Unlock()
		return
	}
	n.px, n.py = x, y
	if w > 0 && h > 0 {
		n.rw, n.rh = model.ClampW(w), model.ClampH(h)
	}
	c.mu.Unlock()
	c.pushPass(n)
}
```

```go
// NoteMoved 用户拖动结束（前端上报逻辑坐标）。
func (c *Controller) NoteMoved(id string, x, y int) {
	c.mu.Lock()
	n := c.findLocked(id)
	if n == nil {
		c.mu.Unlock()
		return
	}
	n.rec.X, n.rec.Y = x, y
	n.px, n.py = x, y
	n.rec.UpdatedAt = time.Now().UnixMilli()
	c.mu.Unlock()
	c.pushPass(n) // 兜底弹开：松手残留重叠 / 拖动中停顿记账 / 键盘移动（无会话事件）
	c.scheduleSave()
}
```

```go
// NoteResized 用户缩放结束（前端上报逻辑尺寸）。
func (c *Controller) NoteResized(id string, w, h int) {
	c.mu.Lock()
	n := c.findLocked(id)
	if n == nil {
		c.mu.Unlock()
		return
	}
	n.rec.W, n.rec.H = model.ClampW(w), model.ClampH(h)
	n.rw, n.rh = 0, 0 // 会话实时尺寸清零，后续几何计算回退 rec 值
	n.rec.UpdatedAt = time.Now().UnixMilli()
	c.mu.Unlock()
	c.pushPass(n) // 兜底弹开：缩放变大压到兄弟
	c.scheduleSave()
```

```go
// startDragSession 拖动/缩放手势采样（v3.1.0 实时挤开）：每 50ms 读窗口实时
// 几何交控制器 DragLiveTick。同窗口重复 Start 取旧起新；End 或窗口消失即退。
// 读窗口几何在锁外、goroutine 内进行，不与控制器锁叠加。
func (o *wailsOps) startDragSession(id string) {
	o.animMu.Lock()
	if done, ok := o.dragSessions[id]; ok {
		close(done)
	}
	done := make(chan struct{})
	o.dragSessions[id] = done
	o.animMu.Unlock()

	go func() {
		ticker := time.NewTicker(dragTickMs * time.Millisecond)
		defer ticker.Stop()
		for {
			select {
			case <-done:
				o.finishDragSession(id, done)
				return
			case <-ticker.C:
				win := o.lookup(id)
				if win == nil {
					o.finishDragSession(id, done)
					return
				}
				mx, my := win.Position()
				mw, mh := win.Size()
				o.ctl.DragLiveTick(id, mx, my, mw, mh)
			}
		}
	}()
}

// stopDragSession 手势结束（EndMove/EndResize）：停当前会话。
func (o *wailsOps) stopDragSession(id string) {
	o.animMu.Lock()
	if done, ok := o.dragSessions[id]; ok {
		delete(o.dragSessions, id)
		close(done)
	}
	o.animMu.Unlock()
}

func (o *wailsOps) finishDragSession(id string, done chan struct{}) {
	o.animMu.Lock()
	if o.dragSessions[id] == done { // 仍是本会话持有标记才清理（被新会话接管则不动）
		delete(o.dragSessions, id)
	}
	o.animMu.Unlock()
}
```
