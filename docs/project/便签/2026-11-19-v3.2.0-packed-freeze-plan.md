> **修订记录**：本计划为首版「让位预算」计划（`2026-11-19-v3.2.0-push-budget-plan.md`，已删除）经开发者验收否决后的修正版——预算把空闲桌面应有的连续挤开也禁掉了；修正为满屏熔断。

**Goal:** 拖拽挤开加「满屏熔断」——桌面空闲时无限实时挤开（v3.1.0 手感），手势内首次出现「四向弹开+装箱兜底全部无解」即熔断本手势（此后接受一切重叠、立即静止）。

**Architecture:** `ResolveOverlaps` 返回 `(pushes, packed)`（接受重叠分支置 packed=true，即「桌面塞不下」的结构性见证）；`Controller.dragPacked` 手势标记，`pushPass(self, trackPacked)` 先查标记再解析、解析后置位；`BeginDragSession` 手势开始重置；召唤路径 `trackPacked=false` 不读不写。

**Tech Stack:** Go（Wails v3 后端）、go test 单测（fakeOps 内存替身，屏 1920x1080；测试尺寸须 ≥ Clamp 最小尺寸）。

**Spec:** `docs/project/便签/2026-11-19-v3.2.0-packed-freeze-design.md`

// 屏幕满：接受重叠 → packed=true（桌面塞不下的见证，触发拖拽手势熔断）。
func TestResolveReportsPackedOnAcceptOverlap(t *testing.T) {
	self := Rect{X: 8, Y: 8, W: 784, H: 584}
	notes := []PushNote{{ID: "b", Rect: Rect{X: 100, Y: 100, W: 100, H: 100}}}

	if len(got) != 0 || !packed {
		t.Fatalf("接受重叠应报告满屏: got %+v packed=%v", got, packed)
	}
}

	notes := []PushNote{{ID: "b", Rect: Rect{X: 58, Y: 58, W: 100, H: 100}}}
	got, packed := ResolveOverlaps(self, notes, 1920, 1080)
	if len(got) != 1 || packed {
		t.Fatalf("成功弹开不应报告满屏: got %+v packed=%v", got, packed)
	}
}

// 满屏熔断：首轮 A 压 B（弹开成功）+ 巨型 C 接受重叠（塞不下见证）→ 熔断；
// A 追压 B → 冻结无弹开、B 保持原位。
func TestDragPackedModeFreezes(t *testing.T) {
	a := model.NewRecord("", 400, 300, true)
	a.Slot = 1
	a.X, a.Y = 8, 8
	b := model.NewRecord("", 240, 220, true)
	b.Slot = 2
	b.X, b.Y = 250, 150
	c := model.NewRecord("", model.MaxCoord, model.MaxCoord, true)
	c.Slot = 3
	c.X, c.Y = 8, 8
	ctl, ops := newTestController(t, a, b, c)
	idA, idB := ctl.noteIDs()[0], ctl.noteIDs()[1]

	ctl.DragLiveTick(idA, 8, 8, 400, 300)
	// b 右弹 (416,150)（dx166 = dy166 平手偏右）；c 四向+装箱无解 → 接受重叠、熔断
	if len(ops.glides) != 1 || ops.glides[0].id != idB || ops.glides[0].x != 416 || ops.glides[0].y != 150 {
		t.Fatalf("首轮应只弹 b (416,150)，c 接受重叠: %+v", ops.glides)
	}
	ops.glides = nil
	ctl.DragLiveTick(idA, 416, 150, 400, 300) // A 追压 b 新位（满屏模式应冻结）
	if len(ops.glides) != 0 {
		t.Fatalf("满屏熔断后不应再弹开: %+v", ops.glides)
	}
	if x, y := ctl.livePosForTest(idB); x != 416 || y != 150 {
		t.Fatalf("B 应保持原位: %d,%d", x, y)
	}
}

func TestDragPackedModeResetsOnNewSession(t *testing.T) {

	a.Slot = 1
	a.X, a.Y = 8, 8
	b := model.NewRecord("", 240, 220, true)

	ctl.BeginDragSession()                    // 新手势重置

		t.Fatalf("新手势应恢复解析，b 左弹 (168,150): %+v", ops.glides)
	}
}

		return 0

	pushes, packed := ResolveOverlaps(selfRect, notes, sw, sh)
	if len(pushes) == 0 {
		return 0
	}
	for i, p := range pushes {
		...（写实时几何 + placed 补写 rec + GlideNote，同 v3.1.0）...
	}
