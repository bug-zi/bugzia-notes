```go
// picture.go 图片粘贴支持（v4.2.0）：picture 目录的保存、引用提取与启动 GC。
// 设计见 docs/project/便签/archive 机制下的 v4.2.0 spec（归档前在本目录）。
package store

import (
	"crypto/rand"
	"encoding/base64"
	"errors"
	"fmt"
	"math/big"
	"os"
	"path/filepath"
	"regexp"
	"strings"
	"time"
)

// MaxPictureBytes 单张图片字节上限（超限拒绝落盘）。
const MaxPictureBytes = 20 << 20

// pictureNameRe 严格文件名格式：YYYYMMDD-HHMMSS-4位随机小写字母数字.扩展名。
// 宽松白名单（[\w.-]+）会放行「..」使 /picture/.. 指向上级目录，必须严格格式。
var pictureNameRe = regexp.MustCompile(`^\d{8}-\d{6}-[a-z0-9]{4}\.(png|jpg|webp|gif)$`)

// pictureTokenRe 图片引用 token 行（独占一行才识别，提取/GC 共用）。
var pictureTokenRe = regexp.MustCompile(`^!\[img\]\(picture/(\d{8}-\d{6}-[a-z0-9]{4}\.(?:png|jpg|webp|gif))\)$`)

// IsPictureName 判断文件名是否符合严格格式（资产路由与打开事件复核共用）。
func IsPictureName(name string) bool { return pictureNameRe.MatchString(name) }

// NormalizeExt 扩展名白名单归一（jpeg→jpg）；非法返回错误。
func NormalizeExt(ext string) (string, error) {
	switch strings.ToLower(strings.TrimPrefix(ext, ".")) {
	case "png":
		return "png", nil
	case "jpg", "jpeg":
		return "jpg", nil
	case "webp":
		return "webp", nil
	case "gif":
		return "gif", nil
	}
	return "", fmt.Errorf("不支持的图片格式: %q", ext)
}

// SavePicture 解码 base64 图片写入 dir（tmp/rename 原子写），返回生成的文件名。
func SavePicture(dir, ext, b64 string) (string, error) {
	e, err := NormalizeExt(ext)
	if err != nil {
		return "", err
	}
	data, err := base64.StdEncoding.DecodeString(b64)
	if err != nil {
		return "", fmt.Errorf("图片数据解码失败: %w", err)
	}
	if len(data) == 0 {
		return "", errors.New("图片数据为空")
	}
	if len(data) > MaxPictureBytes {
		return "", fmt.Errorf("图片超过 %dMB 上限", MaxPictureBytes>>20)
	}
	name, err := newPictureName(e)
	if err != nil {
		return "", err
	}
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return "", err
	}
	if err := atomicWrite(filepath.Join(dir, name), data); err != nil {
		return "", err
	}
	return name, nil
}

// newPictureName 生成「时间戳-4位随机小写字母数字」文件名。
func newPictureName(ext string) (string, error) {
	const alphabet = "abcdefghijklmnopqrstuvwxyz0123456789"
	suffix := make([]byte, 4)
	for i := range suffix {
		n, err := rand.Int(rand.Reader, big.NewInt(int64(len(alphabet))))
		if err != nil {
			return "", err
		}
		suffix[i] = alphabet[n.Int64()]
	}
	return time.Now().Format("20060102-150405") + "-" + string(suffix) + "." + ext, nil
}

// ExtractPictureRefs 提取内容中的图片文件名引用（token 须独占一行才计入）。
func ExtractPictureRefs(contents []string) map[string]bool {
	refs := map[string]bool{}
	for _, c := range contents {
		for _, line := range strings.Split(c, "\n") {
			if m := pictureTokenRe.FindStringSubmatch(strings.TrimSpace(line)); m != nil {
				refs[m[1]] = true
			}
		}
	}
	return refs
}

// ReferencedPictures 汇总便签与回收站引用的图片名（启动 GC 用；
// LoadNotes 幂等可重复调用，GC 在控制器启动恢复之前执行）。
func ReferencedPictures(st *Store) map[string]bool {
	contents := make([]string, 0, 16)
	for _, n := range st.LoadNotes() {
		contents = append(contents, n.Content)
	}
	for _, e := range st.RecycleEntries() {
		contents = append(contents, e.Content)
	}
	return ExtractPictureRefs(contents)
}

// GCPictures 删除 dir 中未被引用的图片文件，返回删除个数。
// 目录不存在安全返回 0；非严格名文件与子目录一律不动。
func GCPictures(dir string, refs map[string]bool) (int, error) {
	entries, err := os.ReadDir(dir)
	if err != nil {
		if os.IsNotExist(err) {
			return 0, nil
		}
		return 0, err
	}
	removed := 0
	for _, en := range entries {
		if en.IsDir() || !IsPictureName(en.Name()) || refs[en.Name()] {
			continue
		}
		if err := os.Remove(filepath.Join(dir, en.Name())); err != nil {
			return removed, err
		}
		removed++
	}
	return removed, nil
}
```

```go
// pictureHandler 拦截 /picture/<名> 请求从磁盘读图返回（v4.2.0 方案 A）。
// 严格文件名校验防路径穿越（宽松白名单会放行「..」）；图片路径但不合法或
// 不存在直接 404，不落入内嵌资产；其余请求原样转发。
func pictureHandler(dir string, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if strings.HasPrefix(r.URL.Path, "/picture/") {
			name := strings.TrimPrefix(r.URL.Path, "/picture/")
			if store.IsPictureName(name) {
				p := filepath.Join(dir, name)
				if _, err := os.Stat(p); err == nil {
					http.ServeFile(w, r, p)
					return
				}
			}
			http.NotFound(w, r)
			return
		}
		next.ServeHTTP(w, r)
	})
}
```

```js
function render() {
  if (editing) return;
  textEl.classList.toggle("md", mdOn);
  textEl.textContent = "";
  // 按行混排（v4.2.0）：图片 token 行 → <img>；连续其余行合段渲染
  // （mdOn 段走 mdRender 富文本，否则纯文本 div——pre-wrap 继承，视觉不变）。
  // 纯文本 div 带 data-segstart（段首在 content 中的字符下标），供
  // caretIndexAtClick 把点击位置映射回 textarea 光标；md 段无下标映射（同前）。
  const lines = content.split("\n");
  let seg = [];
  let segStart = 0;
  let cur = 0;
  const flush = () => {
    const text = seg.join("\n");
    if (text.trim() !== "") {
      if (mdOn) {
        const box = document.createElement("div");
        box.innerHTML = mdRender(text);
        textEl.appendChild(box);
      } else {
        const div = document.createElement("div");
        div.textContent = text;
        div.dataset.segstart = String(segStart);
        textEl.appendChild(div);
      }
    }
    seg = [];
  };
  for (const line of lines) {
    const m = PICTURE_TOKEN.exec(line.trim());
    if (m) {
      flush();
      textEl.appendChild(imgEl(m[1]));
    } else {
      if (seg.length === 0) segStart = cur;
      seg.push(line);
    }
    cur += line.length + 1; // +1 = 换行符
  }
  flush();
}

/* 图片缩略元素：点击转发 Go 用系统默认查看器打开；加载失败（文件被 GC/损坏）
   就地替换为占位文本行。 */
function imgEl(name) {
  const img = document.createElement("img");
  img.className = "note-img";
  img.src = "/picture/" + encodeURIComponent(name);
  img.alt = "图片 " + name;
  img.addEventListener("click", (e) => {
    e.stopPropagation(); // 防冒泡进 bodyEl 的「点击进编辑」
    wails.Events.Emit("note:openimage", { id, name });
  });
  img.addEventListener("error", () => {
    const div = document.createElement("div");
    div.className = "note-img-missing";
    div.textContent = "[图片缺失 " + name + "]";
    img.replaceWith(div);
  });
  return img;
}
```

```js
/* 粘贴图片（v4.2.0，仅辅助/自由便签）：clipboardData 含 image/* 时拦截，
   转 base64 交 Go 落盘 picture 目录，成功后在光标处插入 token。无图片
   不拦截（放行默认文本粘贴）；主便签 slot=0 无编辑态，天然不触发。 */
const PASTE_EXT = {
  "image/png": "png", "image/jpeg": "jpg",
  "image/webp": "webp", "image/gif": "gif",
};
ta.addEventListener("paste", (e) => {
  if (slot === 0) return;
  const items = e.clipboardData && e.clipboardData.items;
  if (!items) return;
  let file = null;
  let ext = "";
  for (const it of items) {
    if (PASTE_EXT[it.type]) {
      file = it.getAsFile();
      ext = PASTE_EXT[it.type];
      break;
    }
  }
  if (!file) return;
  e.preventDefault();
  const reader = new FileReader();
  reader.onload = () => {
    const b64 = String(reader.result || "").split(",")[1] || "";
    wails.Call.ByName(NS + "SavePicture", ext, b64)
      .then((name) => {
        if (name) insertPictureToken(String(name));
      })
      .catch((err) => console.error("[image] 保存失败:", err));
  };
  reader.onerror = () => console.error("[image] 读取剪贴板图片失败");
  reader.readAsDataURL(file);
});

/* 在 ta 光标处插入图片 token：光标所在行非空则先补换行，token 后补换行
   （保证独占一行）。停留在编辑态走既有失焦/Enter 提交通道（连续贴多张
   不打断输入）；程序化赋值不触发 input 事件，draft 须手动同步。 */
function insertPictureToken(name) {
  const token = "![img](picture/" + name + ")";
  const pos = ta.selectionStart ?? ta.value.length;
  const end = ta.selectionEnd ?? pos;
  const before = ta.value.slice(0, pos);
  const after = ta.value.slice(end);
  const pre = before !== "" && !before.endsWith("\n") ? "\n" : "";
  const next = before + pre + token + "\n" + after;
  ta.value = next;
  draft = next;
  const caret = (before + pre + token + "\n").length;
  ta.setSelectionRange(caret, caret);
  syncTextareaHeight();
}
```
