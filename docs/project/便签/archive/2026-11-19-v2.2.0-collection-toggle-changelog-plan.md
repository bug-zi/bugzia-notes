- [ ] **Step 3: 实现 ToggleMainSet**

`internal/wm/controller.go` 的 `createFreeNote` 函数后插入：

```go
// ToggleMainSet 「便签集合切换」键（toggle）入口（v2.2.0）：集合 = 主便签 +
// 全部自由便签，辅助便签不参与。展开 = 集合内每张钉住且可见（主便签缺失即
// 未展开）；未展开 → 补建缺失的主便签、召唤全部隐藏成员、未钉成员升钉保证
// 可见（防空转，升钉不刷 UpdatedAt）、聚焦集合内最近活跃一张；已展开 →
// 集合全部收起。
func (c *Controller) ToggleMainSet() {
	c.mu.Lock()
	var main *NoteLive
	set := make([]*NoteLive, 0, len(c.notes))
	for _, n := range c.notes {
		switch n.rec.Slot {
		case model.SlotMain:
			main = n
			set = append(set, n)
		case model.SlotFree:
			set = append(set, n)
		}
	}
	expanded := main != nil
	for _, n := range set {
		if !n.rec.Pinned || !c.visible[n.rec.ID] {
			expanded = false
			break
		}
	}
	if expanded {
		ids := make([]string, 0, len(set))
		for _, n := range set {
			ids = append(ids, n.rec.ID)
		}
		c.mu.Unlock()
		for _, id := range ids {
			c.hideNote(id)
		}
		return
	}

	// 召唤分支：聚焦目标先定（UpdatedAt 最大；补建场景聚焦交给新主便签的
	// needsFocus 机制，不在此处）
	focusID := ""
	var focusAt int64
	if main != nil {
		for _, n := range set {
			if focusID == "" || n.rec.UpdatedAt > focusAt {
				focusID, focusAt = n.rec.ID, n.rec.UpdatedAt
			}
		}
	}
	promoted := make([]string, 0)
	summoned := make([]string, 0)
	for _, n := range set {
		if !n.rec.Pinned {
			n.rec.Pinned = true // 仅改层级，不刷 UpdatedAt
			promoted = append(promoted, n.rec.ID)
		}
		if !c.visible[n.rec.ID] {
			summoned = append(summoned, n.rec.ID)
		}
	}
	c.mu.Unlock()

	if main == nil {
		c.createSlotNote(model.SlotMain, false) // 补建：即钉住、装箱、needsFocus
	}
	for _, id := range summoned {
		c.showNote(id)
	}
	for _, id := range promoted {
		c.ops.SetPinned(id, true)
	}
	if len(promoted) > 0 || main == nil {
		c.scheduleSave() // 升钉 / 补建一并固化（防抖，非同步写盘）
	}
	if focusID != "" {
		c.ops.FocusNote(focusID)
		c.mu.Lock()
		c.activeID = focusID // 召唤显示即成为当前便签
		c.mu.Unlock()
	}
}
```

/* ── 更新日志（内置静态数据 changelog.js，当前版本高亮）────────────── */

const changelogBox = document.getElementById("changelog");
document.getElementById("show-changelog").addEventListener("click", () => {
  if (!changelogBox.hidden) {
    changelogBox.hidden = true;
    return;
  }
  renderChangelog();
  changelogBox.hidden = false;
});

function renderChangelog() {
  changelogBox.textContent = "";
  for (const entry of CHANGELOG) {
    const head = document.createElement("p");
    head.className = "changelog-ver";
    const tag = document.createElement("strong");
    tag.textContent = entry.version;
    head.appendChild(tag);
    const date = document.createElement("span");
    date.className = "changelog-date";
    date.textContent = "　" + entry.date;
    head.appendChild(date);
    if (entry.version === currentVersion) {
      const cur = document.createElement("span");
      cur.className = "changelog-cur";
      cur.textContent = "当前版本";
      head.appendChild(cur);
    }
    changelogBox.appendChild(head);
    const ul = document.createElement("ul");
    for (const item of entry.items) {
      const li = document.createElement("li");
      li.textContent = item;
      ul.appendChild(li);
    }
    changelogBox.appendChild(ul);
  }
}
