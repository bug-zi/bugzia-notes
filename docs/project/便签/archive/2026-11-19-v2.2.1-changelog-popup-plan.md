// openChangelog 打开更新日志窗口（单例，窗体参数对齐回收站窗口）。
func openChangelog(app *application.App) {
	if win := lookupWindow(app, "changelog"); win != nil {
		win.Show()
		win.Focus()
		return
	}
	app.Window.NewWithOptions(application.WebviewWindowOptions{
		Name:            "changelog",
		Title:           "bugzia-notes 更新日志",
		URL:             "/changelog.html",
		Width:           460,
		Height:          600,
		DevToolsEnabled: true,
	})
}
```

/* 渲染更新日志列表（changelog.html 消费；currentVersion 来自服务 Version）。 */
export function renderChangelog(box, currentVersion) {
  box.textContent = "";
  for (const entry of CHANGELOG) {
    const head = document.createElement("p");
    head.className = "changelog-ver";
    const tag = document.createElement("strong");
    tag.textContent = entry.version;
    head.appendChild(tag);
    const date = document.createElement("span");
    date.className = "changelog-date";
    date.textContent = "　" + entry.date;
    head.appendChild(date);
    if (entry.version === currentVersion) {
      const cur = document.createElement("span");
      cur.className = "changelog-cur";
      cur.textContent = "当前版本";
      head.appendChild(cur);
    }
    box.appendChild(head);
    const ul = document.createElement("ul");
    for (const item of entry.items) {
      const li = document.createElement("li");
      li.textContent = item;
      ul.appendChild(li);
    }
    box.appendChild(ul);
  }
}
```
