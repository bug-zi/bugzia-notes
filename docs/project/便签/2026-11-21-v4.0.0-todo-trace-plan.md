// TodoHeaderName 识别「【分类名】」标题行：整行恰为该形态才返回名称与 true
//（调用方再校验名称是否在已配置分类中，防止待办正文恰似【x】被误判）。
func TodoHeaderName(line string) (string, bool) {
	t := strings.TrimSpace(line)
	if !strings.HasPrefix(t, "【") || !strings.HasSuffix(t, "】") {
		return "", false
	}
	name := strings.TrimSuffix(strings.TrimPrefix(t, "【"), "】")
	if strings.TrimSpace(name) == "" {
		return "", false
	}
	return name, true
}

// NormalizeTodoCategories 规范化分类配置：trim、rune 截 MaxTitleLen、去空白项、
// 去重（保留首个）、「未分类」强制存在且恒排末尾。
func NormalizeTodoCategories(cats []string) []string {
	out := []string{}
	seen := map[string]bool{}
	for _, c := range cats {
		c = strings.TrimSpace(c)
		if runes := []rune(c); len(runes) > MaxTitleLen {
			c = string(runes[:MaxTitleLen])
		}
		if c == "" || seen[c] || c == TodoUnsorted {
			continue
		}
		seen[c] = true
		out = append(out, c)
	}
	return append(out, TodoUnsorted)
}

// InsertTodoCategories 把 found 中未配置的名插入「未分类」之前（v4 迁移保底用，
// 幂等：已在配置中的不重复插）。
func InsertTodoCategories(cats, found []string) []string {
	known := map[string]bool{}
	for _, c := range cats {
		known[c] = true
	}
	add := []string{}
	for _, f := range found {
		if !known[f] && f != TodoUnsorted {
			known[f] = true
			add = append(add, f)
		}
	}
	if len(add) == 0 {
		return cats
	}
	out := []string{}
	for _, c := range cats {
		if c == TodoUnsorted {
			out = append(out, add...)
		}
		out = append(out, c)
	}
	return out
}

// isDefaultMainTitle 判断标题是否为默认样式「便签n」（v4 迁移判定，自定义名不动）。
func isDefaultMainTitle(t string) bool {
	const prefix = "便签"
	if !strings.HasPrefix(t, prefix) || len(t) <= len(prefix) {
		return false
	}
	for _, r := range t[len(prefix):] {
		if r < '0' || r > '9' {
			return false
		}
	}
	return true
}

// migrateV4 v4.0.0 首启迁移（幂等）：主便签默认标题「便签n」→「待办事项」；
// 主便签 content 中未配置的历史「【名】」标题行自动补进 todo_categories
//（未分类之前，防历史标题被当普通待办）。就地重写 notes.json 与 settings.json
//（失败仅记日志，内存已生效，后续保存补写）。
func (s *Store) migrateV4(notes []model.NoteRecord) []model.NoteRecord {
	changed := false
	for i := range notes {
		if notes[i].Slot != model.SlotMain {
			continue
		}
		if isDefaultMainTitle(notes[i].Title) {
			notes[i].Title = model.MainNoteTitle
			changed = true
		}
		cur := s.Settings()
		known := map[string]bool{}
		for _, c := range cur.TodoCategories {
			known[c] = true
		}

package wm

import (
	"strings"

	"bugzia-notes/internal/model"
)

// TodoSection 主便签分区：分类名 + 该分区下的待办条目（v4.0.0）。
// 一条待办 = 一行文本（条目内部无换行）；「【已配置分类名】」行 = 分区标题。
type TodoSection struct {
	Category string
	Items    []string
}

// ParseTodoContent 把主便签 content 按行解析为分区。仅「【名】且名 ∈ cats」
// 视为分区标题（model.TodoHeaderName + 配置校验双层过滤，防待办正文恰似
//【x】被误判）；标题前的条目与无标题内容归「未分类」；空行丢弃、条目 trim；
// 输出按 cats 顺序，「未分类」恒在末尾（缺失的分区输出空条目表）。
func ParseTodoContent(content string, cats []string) []TodoSection {
	inCats := map[string]bool{}
	for _, c := range cats {
		inCats[c] = true
	}
	buckets := map[string][]string{}
	cur := model.TodoUnsorted
	for _, line := range strings.Split(content, "\n") {
		t := strings.TrimSpace(line)
		if t == "" {
			continue
		}
		if name, ok := model.TodoHeaderName(t); ok && inCats[name] {
			cur = name
			continue
		}
		if !inCats[cur] {
			cur = model.TodoUnsorted
		}
		buckets[cur] = append(buckets[cur], t)
	}
	seen := map[string]bool{}
	out := make([]TodoSection, 0, len(cats)+1)
	for _, c := range cats {
		if c == model.TodoUnsorted || seen[c] {
			continue
		}
		seen[c] = true
		out = append(out, TodoSection{Category: c, Items: buckets[c]})
	}
	return append(out, TodoSection{Category: model.TodoUnsorted, Items: buckets[model.TodoUnsorted]})
}

// RenderTodoContent 分区序列化回 content：分区标题恒写出（空分区由此在内容
// 中显式存在，spec §3.1「分区按配置渲染」的存储形态）。与 ParseTodoContent 互逆。
func RenderTodoContent(secs []TodoSection, cats []string) string {
	byName := map[string][]string{}
	for _, s := range secs {
		byName[s.Category] = s.Items
	}
	lines := make([]string, 0, 16)
	for _, c := range cats {
		lines = append(lines, "【"+c+"】")
		for _, it := range byName[c] {
			lines = append(lines, it)
		}
	}
	return strings.Join(lines, "\n")
}

// RemoveTodoItem 从指定分区移除首个与 text 相同的条目；返回是否移除
//（同分区内同名多条只移除第一条，spec §3.3）。
func RemoveTodoItem(secs []TodoSection, category, text string) bool {
	for i := range secs {
		if secs[i].Category != category {
			continue
		}
		for j, it := range secs[i].Items {
			if it == text {
				secs[i].Items = append(secs[i].Items[:j], secs[i].Items[j+1:]...)
				return true
			}
		}
		return false
	}
	return false
}

// AppendTodoItem 条目追加到指定分区末尾（分区由 ParseTodoContent 保证存在）。
func AppendTodoItem(secs []TodoSection, category, text string) {
	for i := range secs {
		if secs[i].Category == category {
			secs[i].Items = append(secs[i].Items, text)
			return
		}
	}
}

func (c *Controller) CheckTodo(id, category, text string) {

	n.rec.UpdatedAt = now
	c.mu.Unlock()

	c.HydrateTo(id)
	c.ops.EmitDoneChanged()

// ApplyTodoCategories 应用分类配置（主便签设置面板入口）：规范化校验 → 主便签
// 分区按新配置重排（rewrites 为旧名→目标名映射：改名/合并/删除；新增与排序传
// nil——重排时旧分区按原汇入，未在 rewrites 且已消失的分类其待办归未分类）→
// 留痕记录分类名同步改写（done.json 立即写盘）→ 设置落盘 + 防抖保存 notes +
// 主便签单窗重推 + 广播 done:changed。UI 契约：单次操作只传一层映射（值即
// 最终目标名，不做链式解析）。
func (c *Controller) ApplyTodoCategories(cats []string, rewrites map[string]string) error {
	cats = model.NormalizeTodoCategories(cats)
	c.mu.Lock()
	var main *NoteLive
	for _, n := range c.notes {
		if n.rec.Slot == model.SlotMain {
			main = n
			break
		}
	}
	if main == nil {
		c.mu.Unlock()
		return fmt.Errorf("主便签不存在")
	}
	oldCats := c.store.Settings().TodoCategories
	secs := ParseTodoContent(main.rec.Content, oldCats)
	merged := map[string][]string{}
	for _, s := range secs {
		t := s.Category
		if v, ok := rewrites[t]; ok {
			t = v
		}
		if !containsStr(cats, t) {
			t = model.TodoUnsorted
		}
		merged[t] = append(merged[t], s.Items...)
	}
	out := make([]TodoSection, 0, len(cats))
	for _, cat := range cats {
		out = append(out, TodoSection{Category: cat, Items: merged[cat]})
	}
	main.rec.Content = RenderTodoContent(out, cats)
	main.rec.UpdatedAt = time.Now().UnixMilli()
	id := main.rec.ID
	c.mu.Unlock()

	cur := c.store.Settings()
	cur.TodoCategories = cats
	if err := c.store.SaveSettingsNow(cur); err != nil {
		return err
	}
	c.store.RewriteDoneCategories(rewrites)
	c.scheduleSave()
	c.HydrateTo(id)
	c.ops.EmitDoneChanged()
	return nil
}
```

import * as wails from "/wails/runtime.js";

/* Material Symbols：check_box_outline_blank / add
   （fonts.google.com/icons，Apache-2.0），inline SVG 同 note.js 锁图标方式。 */
const PATH_UNCHECKED =
  "M19 3H5c-1.1 0-2 .9-2 2v14c0 1.1.9 2 2 2h14c1.1 0 2-.9 2-2V5c0-1.1-.9-2-2-2zm0 16H5V5h14v14z";
const PATH_ADD =
  "M19 13h-6v6h-2v-6H5v-2h6V5h2v6h6v2z";

const UNSORTED = "未分类";

function svg(path) {
  return '<svg viewBox="0 0 24 24" width="15" height="15" aria-hidden="true"><path d="' + path + '" fill="currentColor"/></svg>';
}

export function mountTodoView({ bodyEl, id, onContent }) {
  const box = document.createElement("div");
  box.className = "note-todos";
  box.hidden = true;
  bodyEl.appendChild(box);

  let sections = [];     // [{category, items}] 渲染与重排的唯一状态源
  let lastContent = "";  // 最近一次注水/提交的内容（避免无变化重复发事件）
  let lastCats = [];     // 最近一次注水的分类配置
  let editing = null;    // {cat, idx, mode: "edit"|"insert", prev, ta}

  /* 与 Go ParseTodoContent 同构 */
  function parse(content, cats) {
    const known = new Set(cats);
    const map = new Map();
    for (const c of cats) map.set(c, []);
    if (!map.has(UNSORTED)) map.set(UNSORTED, []);
    let cur = UNSORTED;
    for (const raw of String(content).split("\n")) {
      const t = raw.trim();
      if (t === "") continue;
      const m = /^【(.+)】$/.exec(t);
      if (m && known.has(m[1])) { cur = m[1]; continue; }
      if (!map.has(cur)) cur = UNSORTED;
      map.get(cur).push(t);
    }
    const secs = [];
    for (const c of cats) {
      if (c !== UNSORTED && !secs.some((s) => s.category === c)) {
        secs.push({ category: c, items: map.get(c) });
      }
    }
    secs.push({ category: UNSORTED, items: map.get(UNSORTED) });
    return secs;
  }

  /* 与 Go RenderTodoContent 同构 */
  function compose(secs) {
    return secs
      .map((s) => ["【" + s.category + "】"].concat(s.items).join("\n"))
      .join("\n");
  }

  function rerender() {
    box.textContent = "";
    for (const sec of sections) box.appendChild(sectionEl(sec));
  }

  function render(content, cats) {
    if (editing) cancelEdit();
    lastContent = content;
    lastCats = cats;
    sections = parse(content, cats);
    rerender();
  }

  function sectionEl(sec) {
    const secEl = document.createElement("div");
    secEl.className = "todo-section";
    const head = document.createElement("div");
    head.className = "todo-head";
    const name = document.createElement("span");
    name.className = "todo-head-name";
    name.textContent = "【" + sec.category + "】";
    const add = document.createElement("button");
    add.type = "button";
    add.className = "todo-add";
    add.title = "添加待办";
    add.innerHTML = svg(PATH_ADD);
    add.addEventListener("click", () => startInsert(sec.category, sec.items.length));
    head.append(name, add);
    secEl.appendChild(head);
    sec.items.forEach((text, idx) => secEl.appendChild(itemEl(sec.category, idx, text)));
    return secEl;
  }

  function itemEl(cat, idx, text) {
    const row = document.createElement("div");
    row.className = "todo-item";
    const chk = document.createElement("button");
    chk.type = "button";
    chk.className = "todo-check";
    chk.title = "完成（记入记事簿）";
    chk.innerHTML = svg(PATH_UNCHECKED);
    chk.addEventListener("click", () => {
      wails.Events.Emit("note:todo-check", { id, category: cat, text });
    });
    const span = document.createElement("span");
    span.className = "todo-text";
    span.textContent = text;
    span.addEventListener("click", () => startEdit(cat, idx));
    row.append(chk, span);
    return row;
  }

  function findSectionEl(cat) {
    const i = sections.findIndex((s) => s.category === cat);
    return i >= 0 ? box.children[i] : null;
  }

  function findRow(cat, idx) {
    const secEl = findSectionEl(cat);
    return secEl ? secEl.querySelectorAll(".todo-item")[idx] || null : null;
  }

  /* ── 行内编辑 ──────────────────────────────────────────────────────
     edit 模式：编辑既有条目（Esc 还原 prev）；insert 模式：在 idx 处插入。
     Enter 提交并在同位置继续录入（粘贴多行按行拆为多条）；空提交不产生条目；
     失焦提交。提交后本地重排 content 走 note:changed。 */

  function syncTaHeight() {
    const ta = editing && editing.ta;
    if (!ta) return;
    ta.style.height = "auto";
    ta.style.height = ta.scrollHeight + "px";
  }

  function bindTa(ta, taCommit) {
    ta.addEventListener("input", syncTaHeight);
    ta.addEventListener("keydown", (e) => {
      if (e.key === "Enter" && !e.isComposing) {
        e.preventDefault();
        taCommit(true);
      } else if (e.key === "Escape") {
        e.preventDefault();
        cancelEdit();
        rerender();
      }
    });
    ta.addEventListener("blur", () => {
      if (editing && editing.ta === ta) taCommit(false);
    });
  }

  function taCommit(continueChain) {
    const ed = editing;
    if (!ed) return;
    editing = null; // 先清（防 remove 触发 blur 重入）
    const parts = ed.ta.value.split("\n").map((s) => s.trim()).filter((s) => s !== "");
    const sec = sections.find((s) => s.category === ed.cat);
    if (ed.mode === "edit") {
      sec.items.splice(ed.idx, 1, ...parts);
    } else {
      sec.items.splice(ed.idx, 0, ...parts);
    }
    const next = compose(sections);
    ed.ta.remove();
    if (next !== lastContent) {
      lastContent = next;
      if (onContent) onContent(next);
      wails.Events.Emit("note:changed", { id, content: next });
    }
    rerender();
    if (continueChain) {
      startInsert(ed.cat, ed.idx + parts.length); // Enter 连续录入
    }
  }

  function cancelEdit() {
    if (!editing) return;
    editing = null;
  }

  function startEdit(cat, idx) {
    if (editing) taCommit(false);
    const row = findRow(cat, idx);
    if (!row) return;
    const ta = document.createElement("textarea");
    ta.className = "todo-edit";
    ta.value = sections.find((s) => s.category === cat).items[idx];
    ta.rows = 1;
    row.querySelector(".todo-text").hidden = true;
    row.appendChild(ta);
    editing = { cat, idx, mode: "edit", prev: ta.value, ta };
    syncTaHeight();
    ta.focus();
    ta.setSelectionRange(ta.value.length, ta.value.length);
    bindTa(ta, taCommit);
  }

  function startInsert(cat, idx) {
    if (editing) taCommit(false);
    const secEl = findSectionEl(cat);
    if (!secEl) return;
    const ta = document.createElement("textarea");
    ta.className = "todo-edit";
    ta.value = "";
    ta.rows = 1;
    ta.placeholder = "输入待办，Enter 继续，空 Enter 结束";
    const rows = secEl.querySelectorAll(".todo-item");
    if (idx < rows.length) {
      secEl.insertBefore(ta, rows[idx]);
    } else {
      secEl.appendChild(ta);
    }
    editing = { cat, idx, mode: "insert", prev: "", ta };
    syncTaHeight();
    ta.focus();
    bindTa(ta, taCommit);
  }

  /* 点分区尾部空白 = 给该分区加待办；点容器空白 = 给最后一个分区加 */
  bodyEl.addEventListener("click", (e) => {
    if (box.hidden) return;
    const t = e.target;
    if (t === bodyEl || t === box) {
      const last = sections[sections.length - 1];
      if (last) startInsert(last.category, last.items.length);
      return;
    }
    if (t.classList && t.classList.contains("todo-section")) {
      const i = Array.prototype.indexOf.call(box.children, t);
      const sec = sections[i];
      if (sec) startInsert(sec.category, sec.items.length);
    }
  });

  function show() { box.hidden = false; }
  function hide() {
    box.hidden = true;
    if (editing) { editing.ta.remove(); editing = null; }
  }
  return { show, hide, render };
}

function catMove(i, d) {
  const next = cats.slice();
  const tmp = next[i];
  next[i] = next[i + d];
  next[i + d] = tmp;
  catApply(next, {});

function startCatRename(row, oldName, i) {
  if (row.querySelector("input")) return;
  const input = document.createElement("input");
  input.type = "text";
  input.maxLength = 30;
  input.value = oldName;
  input.className = "note-cat-rename";
  const name = row.querySelector(".note-cat-name");
  name.hidden = true;
  row.insertBefore(input, name);
  input.focus();
  input.select();
  let done = false;
  const finish = (ok) => {
    if (done) return;
    done = true;
    input.remove();
    name.hidden = false;
    if (!ok) return;
    const v = input.value.trim();
    if (v === "" || v === oldName) return;
    const next = cats.slice();
    next[i] = v;
    catApply(next, { [oldName]: v });
  };
  input.addEventListener("keydown", (e) => {
    if (e.key === "Enter" && !e.isComposing) {
      e.preventDefault();
      finish(true);
    } else if (e.key === "Escape") {
      e.preventDefault();
      finish(false);
    }
  });
  input.addEventListener("blur", () => finish(true));
}

async function catApply(nextCats, rewrites) {

}

}
btnCatAdd.addEventListener("click", addCat);
catNewEl.addEventListener("keydown", (e) => {

    addCat();
  }

const btnCat = document.getElementById("v-cat");

let entries = []; // 新到旧
let cats = [];    // 分类配置（组序）
let mode = "time";

function fmtDone(ms) {
  if (!ms) return "";
  const d = new Date(ms);
  const p = (n) => String(n).padStart(2, "0");
  return String(d.getFullYear()).slice(2) + p(d.getMonth() + 1) + p(d.getDate()) +
    "-" + p(d.getHours()) + ":" + p(d.getMinutes());
}

function rowEl(e) {
  const row = document.createElement("div");
  row.className = "row";
  const time = document.createElement("span");
  time.className = "time";
  time.textContent = fmtDone(e.done_at);
  const text = document.createElement("span");
  text.className = "text";
  text.textContent = e.text;
  const restore = document.createElement("button");
  restore.type = "button";
  restore.className = "restore";
  restore.textContent = "还原";
  const back = cats.includes(e.category) ? e.category : "未分类";
  restore.title = "放回主便签（" + back + "）";
  restore.addEventListener("click", async () => {
    try {
      await wails.Call.ByName(NS + "RestoreDone", e.id);
    } catch (err) {
      emptyEl.textContent = "还原失败：" + err;
      emptyEl.hidden = false;
    }
  });
  row.append(time, text, restore);
  return row;
}

function render() {
  listEl.textContent = "";
  if (mode === "cat") {
    const groups = new Map(cats.map((c) => [c, []]));
    for (const e of entries) {
      if (!groups.has(e.category)) groups.get("未分类").push(e);
      else groups.get(e.category).push(e);
    }
    for (const [cat, list] of groups) {
      const head = document.createElement("div");
      head.className = "group";
      head.textContent = "【" + cat + "】";
      listEl.appendChild(head);
      for (const e of list) listEl.appendChild(rowEl(e));
    }
  } else {
    for (const e of entries) listEl.appendChild(rowEl(e));
  }
  emptyEl.hidden = entries.length > 0;
  if (emptyEl.textContent.startsWith("还原失败") || emptyEl.textContent.startsWith("读取失败")) {
    emptyEl.hidden = false;
  }
}

async function load() {
  try {
    const [got, gotCats] = await Promise.all([
      wails.Call.ByName(NS + "ListDone"),
      wails.Call.ByName(NS + "ListTodoCategories"),
    ]);
    entries = Array.isArray(got) ? got : [];
    cats = Array.isArray(gotCats) && gotCats.length > 0 ? gotCats : ["未分类"];
    if (!cats.includes("未分类")) cats.push("未分类");
  } catch (err) {
    entries = [];
    emptyEl.textContent = "读取失败：" + err;
    emptyEl.hidden = false;
  }
  render();
}

btnTime.addEventListener("click", () => {
  mode = "time";
  btnTime.classList.add("active");
  btnCat.classList.remove("active");
  render();
});
btnCat.addEventListener("click", () => {
  mode = "cat";
  btnCat.classList.add("active");
  btnTime.classList.remove("active");
  render();
});

wails.Events.On("done:changed", load);
document.addEventListener("visibilitychange", () => {
  if (!document.hidden) load();
});

load();
```
