```go
// migrateNotesToV3 v1 → v3 升级：最新更新的自由便签升级为主便签（Slot=0，
// 字段全保留），其余不加载；原文件改名备份留存后立即以 v3 落盘
// （v1→v2 原迁移逻辑 + v3 标题，见 2026-11-18-v2.1.0-free-notes-title-design.md §5.1）。
// 改名失败：原文件不动、按空表启动（防半迁移态，下次启动重试）；
// v3 重写失败仅记日志（内存已是 v3 语义，后续防抖保存补写）。
func (s *Store) migrateNotesToV3(notes []model.NoteRecord) []model.NoteRecord {
	if len(notes) == 0 {
		return nil
	}
	backup := s.NotesPath() + ".bak-" + time.Now().Format("20060102-150405")
	if _, err := os.Stat(backup); err == nil {
		backup += "-" + strconv.FormatInt(time.Now().UnixNano(), 10)
	}
	if err := os.Rename(s.NotesPath(), backup); err != nil {
		log.Printf("[store] notes.json v1 备份改名失败，放弃迁移（按空表启动）: %v", err)
		return nil
	}
	log.Printf("[store] 老数据已备份: %s", backup)
	latest := 0
	for i, n := range notes {
		if n.UpdatedAt > notes[latest].UpdatedAt {
			latest = i
		}
	}
	main := notes[latest]
	main.Slot = model.SlotMain
	out := defaultTitles([]model.NoteRecord{main})
	if err := s.writeNotes(out); err != nil {
		log.Printf("[store] notes.json v3 重写失败（内存已生效，稍后防抖补写）: %v", err)
	}
	return out
}
```

4. 文件内追加 helper（`writeNotes` 附近）：

```go
// defaultTitles 为空标题记录按序补默认「便签n」（幂等：已有标题参与占用
// 判定；第 i 个空标题记录从 i+1 起试）。直接改入参切片并返回。
func defaultTitles(notes []model.NoteRecord) []model.NoteRecord {
	taken := map[string]bool{}
	for i := range notes {
		if notes[i].Title != "" {
			taken[notes[i].Title] = true
		}
	}
	for i := range notes {
		if notes[i].Title == "" {
			notes[i].Title = model.NextDefaultTitle(taken, i+1)
			taken[notes[i].Title] = true
		}
	}
	return notes
}
```

// CreateOrSummonFree 「新建便签」键（main）入口：有隐藏自由便签先召唤
// UpdatedAt 最大的一张（静默启动恢复场景），否则新建一张自由便签。
func (c *Controller) CreateOrSummonFree() {
	c.mu.Lock()
	var best *NoteLive
	for _, n := range c.notes {
		if n.rec.Slot != model.SlotFree || c.visible[n.rec.ID] {
			continue
		}
		if best == nil || n.rec.UpdatedAt > best.rec.UpdatedAt {
			best = n
		}
	}
	id := ""
	if best != nil {
		id = best.rec.ID
	}
	c.mu.Unlock()
	if id == "" {
		c.createFreeNote()
		return
	}
	c.showNote(id)
	c.ops.FocusNote(id)
	c.mu.Lock()
	c.activeID = id // 召唤显示即成为当前便签
	c.mu.Unlock()
}

// createFreeNote 新建自由便签：未锁定时跟随全局默认样式（独立样式设置的
// 基线），即钉住、装箱定位、聚焦成为当前便签。
func (c *Controller) createFreeNote() {
	if !c.canAdd() {
		c.Notify("bugzia-notes", fmt.Sprintf("便笺已达上限（%d）", model.MaxNotes))
		return
	}
	s := c.store.Settings()
	rec := model.NewRecord("", s.W, s.H, true)
	rec.Slot = model.SlotFree
	rec.Title = c.nextDefaultTitle()
	live := c.openRecord(rec, true, false)
	c.mu.Lock()
	c.activeID = live.rec.ID
	c.needsFocus[live.rec.ID] = true
	c.mu.Unlock()
}

func (c *Controller) nextDefaultTitle() string {
	c.mu.Lock()

func (c *Controller) SetTitle(id, title string) {

function commitTitle() {
  if (!titleEditing) return;
  titleEditing = false;
  titleInput.hidden = true;
  titleEl.hidden = false;
  const next = titleInput.value;
  if (next !== title && next.trim() !== "") {

}

  titleEl.hidden = false;
